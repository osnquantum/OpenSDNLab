    );

"Create batch first"

});


}

async function checkReadiness(){

    try{

        let r = await fetch("/api/readiness");

        let data = await r.json();

        let box=document.getElementById("readiness");

        if(data.ready){

            box.innerHTML=
            "🟢 SYSTEM READY<br>" +
            "API ✅<br>" +
            "Controller ✅<br>" +
            "OVS ✅<br>" +
            "Mininet ✅<br>" +
            "Database ✅";

        }
        else{

            box.innerHTML=
            "🔴 SYSTEM NOT READY<br>" +
            JSON.stringify(data.checks);

        }

    }

    catch(e){

        document.getElementById("readiness").innerHTML =
        "ERROR CHECKING SYSTEM";

    }

}


checkReadiness();
